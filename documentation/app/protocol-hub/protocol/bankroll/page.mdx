## Research Questions

### 1. What is the current max bet size on most used chains?

Max bet is dynamically calculated based on bankroll size and balance risk:

```
Max Payout = Bankroll Balance × Balance Risk (typically 2%)
Max Bet = Max Payout / (Multiplier - 1)
```

**Example calculations:**

| Bankroll Size | Balance Risk | Max Payout | Game (Multiplier) | Max Bet |
|---------------|--------------|------------|-------------------|---------|
| $10,000 | 2% | $200 | Coin Toss (2x) | $200 |
| $10,000 | 2% | $200 | Dice (10x) | $22.22 |
| $50,000 | 2% | $1,000 | Roulette (36x) | $28.57 |

**To find current max bets:** Check the bankroll balances using links below, then apply the 2% formula.

## Bankroll Query Script

A TypeScript script is available to query current bankroll balances across all chains.

### Setup

```bash
mkdir betswirl-query && cd betswirl-query
npm init -y
npm install viem
```

### Script: `betswirl-bankroll-query.ts`

import { base, arbitrum, polygon, bsc, avalanche } from 'viem/chains';

const ERC20_ABI = [

async function getTokenBalance(
  client: ReturnType<typeof createPublicClient>,
  tokenAddress: Address,
  holderAddress: Address,
  decimals: number
): Promise<bigint> {
  if (tokenAddress === '0x0000000000000000000000000000000000000000') {
    return await client.getBalance({ address: holderAddress });
  } else {
    try {
      return await client.readContract({
        address: tokenAddress,
        abi: ERC20_ABI,
        functionName: 'balanceOf',
        args: [holderAddress],
      });
    } catch (e) {
      return 0n;
    }
  }
}

async function queryChain(chainConfig: ChainConfig) {
  const client = createPublicClient({
    chain: chainConfig.chain,
    transport: http(chainConfig.rpcUrl),
  });

  console.log(`\n${'='.repeat(60)}`);
  console.log(`📍 ${chainConfig.name.toUpperCase()}`);
  console.log(`   Bank: ${chainConfig.explorerUrl}/address/${BANK_ADDRESS}`);
  console.log('='.repeat(60));

  const chainKey = chainConfig.name.toLowerCase().replace(' ', '');
  const tokenMap = chainKey === 'bnbchain' ? TOKENS.bsc : TOKENS[chainKey] || {};

  let totalUsdValue = 0;

  // Get native balance
  const nativeBalance = await client.getBalance({ address: BANK_ADDRESS });
  const nativeFormatted = formatEther(nativeBalance);
  const nativeUsd = parseFloat(nativeFormatted) * (PRICES[chainConfig.nativeSymbol] || 0);
  totalUsdValue += nativeUsd;

  console.log(`\n💰 Native Token (${chainConfig.nativeSymbol}):`);
  console.log(`   Balance: ${parseFloat(nativeFormatted).toFixed(6)} ${chainConfig.nativeSymbol}`);
  console.log(`   USD Value: $${nativeUsd.toFixed(2)}`);

  // Get token balances
  console.log(`\n🪙 Token Balances:`);
  for (const [symbol, token] of Object.entries(tokenMap)) {
    if (token.address === '0x0000000000000000000000000000000000000000') continue;

    try {
      const balance = await getTokenBalance(client, token.address, BANK_ADDRESS, token.decimals);
      const formatted = formatUnits(balance, token.decimals);
      const usdValue = parseFloat(formatted) * (PRICES[symbol] || 0);

      if (parseFloat(formatted) > 0) {
        totalUsdValue += usdValue;
        console.log(`   ${symbol}: ${parseFloat(formatted).toLocaleString()} (~$${usdValue.toFixed(2)})`);
      }
    } catch (e) {
      // Token not available
    }
  }

  console.log(`\n💵 TOTAL BANKROLL: $${totalUsdValue.toFixed(2)}`);

  // Calculate max payout based on 2% balance risk
  const maxPayout = totalUsdValue * 0.02;
  console.log(`   Max Payout (2% risk): $${maxPayout.toFixed(2)}`);

  return { chain: chainConfig.name, totalUsd: totalUsdValue, maxPayout };
}

async function main() {
  console.log('╔════════════════════════════════════════════════════════════╗');
  console.log('║         BETSWIRL BANKROLL QUERY - ALL CHAINS               ║');
  console.log('║         Bank Contract: 0x8FB3110015FBCAA469...             ║');
  console.log('╚════════════════════════════════════════════════════════════╝');
  console.log(`\n⏰ Timestamp: ${new Date().toISOString()}`);
  console.log(`\n⚠️  Note: USD values use approximate prices. Update PRICES object for accuracy.`);

  const results: { chain: string; totalUsd: number; maxPayout: number }[] = [];

  for (const chainConfig of CHAINS) {
    try {
      const result = await queryChain(chainConfig);
      results.push(result);
    } catch (error) {
      console.log(`\n❌ Error querying ${chainConfig.name}:`, error);
      results.push({ chain: chainConfig.name, totalUsd: 0, maxPayout: 0 });
    }
  }

  // Summary
  console.log(`\n${'═'.repeat(60)}`);
  console.log('📈 CROSS-CHAIN SUMMARY');
  console.log('═'.repeat(60));

  const totalAll = results.reduce((sum, r) => sum + r.totalUsd, 0);

  console.log('\n| Chain       | Bankroll (USD) | Max Payout (2%) |');
  console.log('|-------------|----------------|-----------------|');
  for (const r of results) {
    console.log(`| ${r.chain.padEnd(11)} | $${r.totalUsd.toFixed(2).padStart(12)} | $${r.maxPayout.toFixed(2).padStart(13)} |`);
  }
  console.log('|-------------|----------------|-----------------|');
  console.log(`| TOTAL       | $${totalAll.toFixed(2).padStart(12)} | $${(totalAll * 0.02).toFixed(2).padStart(13)} |`);

  console.log(`\n🔗 Dashboard Links:`);
  console.log(`   Dune Analytics: https://dune.com/betswirl/analytics`);
  console.log(`   DefiLlama: https://defillama.com/protocol/betswirl`);
  console.log(`   BetSwirl App: https://app.betswirl.com/protocol/analytics/affiliate`);
}

main().catch(console.error);
